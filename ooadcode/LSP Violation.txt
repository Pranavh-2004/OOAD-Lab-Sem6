// VIOLATION: Penguin is a Bird but cannot fly.
class Bird 
{
    public void fly() 
    {
        System.out.println("Bird is flying");
    }
}

class Penguin extends Bird 
{
    @Override
    public void fly() 
    {
        // This breaks the program's correctness if a client expects all Birds to fly.
        throw new UnsupportedOperationException("Penguins cannot fly!");
    }
}
